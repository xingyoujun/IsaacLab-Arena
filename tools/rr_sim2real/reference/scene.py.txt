def set_translate(prim, vec):
    """设置平移。XformCommonAPI 遇到 xformOp:orient 会静默失败，故直接操作已有 op。"""
    xf = UsdGeom.Xformable(prim)
    for op in xf.GetOrderedXformOps():
        if op.GetOpType() == UsdGeom.XformOp.TypeTranslate:
            op.Set(vec)
            return
    xf.AddTranslateOp().Set(vec)


def look_at(eye, target, up=Gf.Vec3d(0, 0, 1)):
    """USD 相机朝 -Z 看、+Y 向上；返回相机的世界变换。"""
    view = Gf.Vec3d(*(np.asarray(target, dtype=float) - np.asarray(eye, dtype=float)))
    view.Normalize()
    up_dir = Gf.Vec3d(up)
    up_dir.Normalize()
    # 垂直俯拍时，默认世界 +Z 上方向与视线共线，SetLookAt 会退化。
    # 此时用世界 +Y 定义图像上方，使相机成像平面保持与桌面平行。
    if abs(Gf.Dot(view, up_dir)) > 0.999:
        up_dir = Gf.Vec3d(0, 1, 0)
    m = Gf.Matrix4d()
    m.SetLookAt(Gf.Vec3d(*eye), Gf.Vec3d(*target), up_dir)
    return m.GetInverse()


def build_material(stage):
    """桌面材质，用 UsdPreviewSurface。

    原来这里用 omni.kit.commands 的 CreateMdlMaterialPrim + OmniPBR。USD 层面看
    一切正常（材质 prim 有效、ComputeBoundMaterial 能解析到它、shader input 也
    读得回来），但 RTX 渲染时并没有真正吃到这套 MDL —— 把 metallic 调 0、基色压到
    近黑，渲出来的桌面依然是浅灰的默认材质。UsdPreviewSurface 是 USD 原生的，
    RTX 直接支持，不依赖 MDL 的加载/编译，行为可预期。
    """
    path = "/World/Looks/TableMetal"
    mat = UsdShade.Material.Define(stage, path)
    sh = UsdShade.Shader.Define(stage, path + "/Shader")
    sh.CreateIdAttr("UsdPreviewSurface")
    sh.CreateInput("diffuseColor", Sdf.ValueTypeNames.Color3f).Set(Gf.Vec3f(*TABLE_COLOR))
    sh.CreateInput("metallic", Sdf.ValueTypeNames.Float).Set(TABLE_METALLIC)
    sh.CreateInput("roughness", Sdf.ValueTypeNames.Float).Set(TABLE_ROUGHNESS)
    mat.CreateSurfaceOutput().ConnectToSource(sh.ConnectableAPI(), "surface")
    return mat


def build_table(stage, material):
    """桌子：静态碰撞体（只加 CollisionAPI，不加 RigidBodyAPI）。

    碰撞必须开启 —— 承载物体和阻挡机械臂是同一个物理属性，无法分离。
    """
    leg_h = TABLE_TOP_H - TOP_THICK
    inset = TABLE_SIZE / 2 - LEG / 2 - 0.02
    parts = [("TableTop", (0.0, 0.0, TABLE_TOP_H - TOP_THICK / 2),
              (TABLE_SIZE, TABLE_SIZE, TOP_THICK))]
    for i, (sx, sy) in enumerate([(1, 1), (1, -1), (-1, 1), (-1, -1)]):
        parts.append((f"Leg{i}", (sx * inset, sy * inset, leg_h / 2),
                      (LEG, LEG, leg_h)))

    for name, center, size in parts:
        # 用原生 Cube：size=1 再加 scale，尺寸就是字面值。
        # （isaacsim 的 FixedCuboid 对 scale 有额外换算，实测厚度会被平方）
        cube = UsdGeom.Cube.Define(stage, f"/World/Table/{name}")
        cube.GetSizeAttr().Set(1.0)
        xf = UsdGeom.Xformable(cube.GetPrim())
        xf.AddTranslateOp().Set(Gf.Vec3d(*center))
        xf.AddScaleOp().Set(Gf.Vec3f(*size))
        UsdPhysics.CollisionAPI.Apply(cube.GetPrim())
        # MaterialBindingAPI 是 applied schema，必须先 Apply 再 Bind。
        # 直接 UsdShade.MaterialBindingAPI(prim).Bind(...) 不会真正写入绑定，
        # 桌子会静默地退回默认材质（改颜色/金属度全都看不出效果）。
        UsdShade.MaterialBindingAPI.Apply(cube.GetPrim()).Bind(material)


def build_lights(stage):
    """灯光强度。

    原来这套强度（dome 1200 / key 4500 / fill 1500）严重过曝：把桌面基色压到
    (0.12,0.12,0.13) 渲出来仍是浅灰，纯红测试时红通道饱和在 252、而材质里为 0 的
    绿蓝通道也有 85。等于材质怎么调都被冲掉。这里整体降到约 1/8，深色材质才出得来。
    需要临时增减用 --light-scale。
    """
    k = args.light_scale
    dome = UsdLux.DomeLight.Define(stage, Sdf.Path("/World/Lights/DomeLight"))
    dome.CreateIntensityAttr(145.0 * k)
    dome.CreateColorAttr(Gf.Vec3f(0.85, 0.88, 0.95))

    key = UsdLux.DistantLight.Define(stage, Sdf.Path("/World/Lights/KeyLight"))
    key.CreateIntensityAttr(540.0 * k)
    key.CreateAngleAttr(1.5)
    key.CreateColorAttr(Gf.Vec3f(1.0, 0.97, 0.92))
    UsdGeom.Xformable(key.GetPrim()).AddRotateXYZOp().Set(Gf.Vec3f(-40.0, 0.0, 30.0))

    fill = UsdLux.DistantLight.Define(stage, Sdf.Path("/World/Lights/FillLight"))
    fill.CreateIntensityAttr(180.0 * k)
    fill.CreateAngleAttr(3.0)
    UsdGeom.Xformable(fill.GetPrim()).AddRotateXYZOp().Set(Gf.Vec3f(-20.0, 0.0, -120.0))


def build_camera(stage):
    """RealSense D435 等效相机。位姿放在父 Xform 上，避免被相机包装类覆盖。"""
    rig = UsdGeom.Xform.Define(stage, "/World/CamRig")
    UsdGeom.Xformable(rig.GetPrim()).AddTransformOp().Set(
        look_at(CAM_POS, CAM_LOOK, Gf.Vec3d(*CAM_UP)))

    cam = UsdGeom.Camera.Define(stage, "/World/CamRig/RealSense_D435")
    cam.GetFocalLengthAttr().Set(CAM["focal"])
    cam.GetHorizontalApertureAttr().Set(CAM["h_aperture"])
    cam.GetVerticalApertureAttr().Set(CAM["v_aperture"])
    cam.GetHorizontalApertureOffsetAttr().Set(CAM["h_offset"])
    cam.GetVerticalApertureOffsetAttr().Set(CAM["v_offset"])
    cam.GetClippingRangeAttr().Set(Gf.Vec2f(0.05, 20.0))
    return cam

if args.verify:
    for _ in range(90):
        world.step(render=False)
    back = art.get_joint_positions()[0]
    print("\n关节        记录值        仿真读回      偏差")
    print("-" * 56)
    worst = 0.0
    for name, want in INIT_JOINTS.items():
        if name not in dof:
            print(f"{name:22s} {want:+.6f}   -- 不在 DOF 中 --")
            continue

        d = got - want
        worst = max(worst, abs(d))
        print(f"{name:22s} {want:+.6f}   {got:+.6f}   {d:+.2e}")
    print("-" * 56)
    print(f"最大偏差 {worst:.3e} rad = {worst*57.2958:.5f}°")

    raise SystemExit(0)
